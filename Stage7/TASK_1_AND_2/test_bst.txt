	bst insert(bst h, int key){
		decl
		enddecl
		begin
			write("inside insert");
			if (h == null) then
				write(key);

				write("in if2");
			else

					h.left  = self.insert(h.left, key);
				else 
					if (key > h.a) then
						 h.right = self.insert(h.right, key);
					endif;
				endif;
			endif;

		end

		decl
			int in;
		enddecl
		begin

				write(h.a);

		end

		decl
			int in;

				in=self.postOrderfun(h.right);
