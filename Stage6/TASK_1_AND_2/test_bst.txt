type
info{
	int x;
	int y;
}
bst{
info a;
bst left;
bst right;
}
endtype
decl
int in,opt;
bst insert(bst h, int key);
int inOrder(bst h);
int preOrder(bst h);
int postOrder(bst h);           
enddecl

bst insert(bst h,int key){
decl
enddecl
begin
if(h==null)then
h=alloc();
h.a=alloc();
h.a.x=key;
h.left=null;
h.right=null;
else if(key<h.a)then
h.left=insert(h.left,key);
else if(key>h.a.x)then
	h.right=insert(h.right, key);
endif;
endif;
endif;
return h;
end
}

int inOrder(bst h){
decl
enddecl
begin
if(h!=null)then
in=inOrder(h.left);
write(h.a.x);
in=inOrder(h.right);
endif;
return 1;
end
}

int preOrder(bst h){
decl
enddecl
begin
if(h!=null) then
write(h.a.x);
in=preOrder(h.left);
in=preOrder(h.right);
endif;
return 1;
end
}

int postOrder(bst h){
decl
enddecl
begin
if(h!=null)then
in=postOrder(h.left);
in=postOrder(h.right);
write(h.a.x);
endif;
return 1;
end
}

int main(){
decl

  bst Root;
